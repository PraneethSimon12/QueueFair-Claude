Django==5.1.15

# Signs the HS256 admission passes the booking service verifies. Pinned to the SAME version the

PyJWT==2.13.0

# Gunicorn manages the Uvicorn (ASGI) workers in the Linux image. It is POSIX-only (it imports

# process — and installs it inside the Docker image (Phase 12), the first time this service runs on
# Linux. uvloop/httptools ride in the same way via uvicorn[standard]'s own win32 marker.
gunicorn==23.0.0 ; sys_platform != "win32"
